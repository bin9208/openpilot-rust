cmake_minimum_required(VERSION 3.16)
project(openpilot_image_codecs_sdk C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Pinned decoder SDK currently supports Linux only")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_INSTALL_LIBDIR lib CACHE STRING "" FORCE)
set(UNIT_TEST OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_JPEG TRUE)
set(LIBYUV_VERSION 1924)
include("${CMAKE_CURRENT_LIST_DIR}/dav1d.cmake")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBYUV_SOURCE}" libyuv EXCLUDE_FROM_ALL)
target_include_directories(yuv PUBLIC "${LIBYUV_SOURCE}/include")
add_library(yuv::yuv ALIAS yuv)
set_target_properties(yuv dav1d PROPERTIES AVIF_LOCAL ON)
set(AVIF_CODEC_DAV1D SYSTEM CACHE STRING "" FORCE)
foreach(codec AOM LIBGAV1 RAV1E SVT AVM)
  set(AVIF_CODEC_${codec} OFF CACHE STRING "" FORCE)
endforeach()
set(AVIF_LIBYUV SYSTEM CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AVIF_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
set(AVIF_JPEG OFF CACHE STRING "" FORCE)
set(AVIF_ZLIBPNG OFF CACHE STRING "" FORCE)
add_subdirectory("${LIBAVIF_SOURCE}" avif)
set(BUILD_CODEC OFF CACHE BOOL "" FORCE)
set(BUILD_DOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENJPEG_SOURCE}" openjpeg)
install(TARGETS yuv ARCHIVE DESTINATION lib)

install(DIRECTORY "${LIBYUV_SOURCE}/include/libyuv" DESTINATION include)
install(FILES "${LIBYUV_SOURCE}/include/libyuv.h" DESTINATION include)

include("${CMAKE_CURRENT_LIST_DIR}/tiff.cmake")
